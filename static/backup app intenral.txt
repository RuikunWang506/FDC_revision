def get_sheet_data(input_value):
    creds = service_account.Credentials.from_service_account_file(SERVICE_ACCOUNT_FILE, scopes=SCOPES)
    service = build('sheets', 'v4', credentials=creds)
    sheet = service.spreadsheets()
    result = sheet.values().get(spreadsheetId=SPREADSHEET_ID, range='updater').execute()
    values = result.get('values', [])
    if not values:
        return "No data found in the sheet."
    else:
        headers = values[0]
        matching_rows = []
        for row in values[1:]:
            if len(row) >= 1:
                if any(input_value.lower() in cell.lower() for cell in row[0:1] + row[7:8]):
                    matching_rows.append(row)
        return [headers] + matching_rows

@app.route("/run_fast_script", methods=["POST"])
def run_fast_script():
    try:
        process = subprocess.Popen([sys.executable, os.path.join(os.path.dirname(__file__), 'fast.py')],
                                   stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True)
        output_lines = []
        while True:
            output = process.stdout.readline()
            if output == '' and process.poll() is not None:
                break
            if output:
                output_lines.append(output.strip())
        return_code = process.poll()
        if return_code == 0:
            output = "\n".join(output_lines)
        else:
            output = "\n".join(output_lines) + "\nError: " + process.stderr.read().strip()
        return jsonify(output)
    except Exception as e:
        logger.error("Error running fast script: %s", e)
        return jsonify(str(e))

@app.route("/loris", methods=["GET", "POST"])
@basic_auth.required
def loris():
    filename = None
    error_message = None
    table_data = None
    input_data = request.form.get("input_data")
    sku = request.form.get("sku", "")
    barcode = request.form.get("barcode", "")
    quantity = request.form.get("quantity", "")
    if request.method == "POST":
        if "submit_data" in request.form:
            try:
                filename = process_loris_stickers(input_data)
                if filename is not None:
                    filename = os.path.basename(filename)
                else:
                    error_message = "An error occurred while processing the data."
            except Exception as e:
                error_message = f"An error occurred: {e}"
        elif "show_table" in request.form:
            try:
                table_data = parse_input_data(input_data)
            except Exception as e:
                error_message = f"An error occurred while parsing the data: {e}"
    return render_template("loris.html", filename=filename, error_message=error_message, table_data=table_data, input_data=input_data, sku=sku, barcode=barcode, quantity=quantity)

def parse_input_data(input_data):
    lines = input_data.strip().split('\n')
    if any(keyword in lines[0] for keyword in ["SKU", "Barcode", "Quantity"]):
        lines = lines[1:]
    table_data = []
    for line in lines:
        if line.strip():
            parts = line.split()
            if len(parts) >= 3:
                sku = parts[0]
                barcode = parts[1]
                quantity = ' '.join(parts[2:])
                table_data.append([sku, barcode, quantity])
    return table_data
